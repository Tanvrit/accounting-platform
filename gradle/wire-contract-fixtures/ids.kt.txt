package fixture

// Rule (b) fixture: an `override var id` needs @SerialName on its own line or on an
// annotation-only line directly above it.
@Serializable
data class PinnedInline(
    @SerialName("id") override var id: String = "",
) : BaseDataClass()

@Serializable
data class PinnedAbove(
    @SerialName("id")
    override var id: String = "",
) : BaseDataClass()

@Serializable
data class Bare(
    @SerialName("name") val name: String = "",
    override var id: String = "",
) : BaseDataClass()

@Serializable
data class PinnedOnlyInComment(
    // @SerialName("id") in a comment does not count
    override var id: String = "",
) : BaseDataClass()
